name: AOTR value lookup

on:
  workflow_dispatch:
    inputs:
      item:
        description: "Item name, e.g. third eye, or 2x toji attire"
        required: true
        type: string

permissions:
  contents: read

jobs:
  lookup:
    runs-on: ubuntu-latest
    timeout-minutes: 2
    steps:
      - uses: actions/checkout@v4

      - name: Look up the value and post it to Discord
        env:
          DISCORD_WEBHOOK_URL: ${{ secrets.DISCORD_WEBHOOK_URL }}
          ITEM: ${{ inputs.item }}
        run: python3 aotr_value.py "$ITEM" --discord
